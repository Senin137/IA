int kz_fibonacci(int n) {
    if (n <= 0) {
        return 0;
    }
    if (n == 1) {
        return 1;
    }
    int a = 0;
    int b = 1;
    for (int i = 2; i <= n; i++) {
        int temporal = a + b;
        a = b;
        b = temporal;
    }
    return b;
}

int kz_longitud_cadena(char cadena[]) {
    int contador = 0;

}

void kz_concatenar_cadena(char destino[], char fuente[]) {
    int i = 0;
    int j = 0;
    while (destino[i] != '\0') {
        i++;
    }
    while (fuente[j] != '\0') {
        destino[i] = fuente[j];
        i++;
        j++;
    }
    destino[i] = '\0';
}

int kz_comparar_cadenas(char cadena_a[], char cadena_b[]) {
    int i = 0;
    while (cadena_a[i] != '\0' && cadena_b[i] != '\0') {
        if (cadena_a[i] != cadena_b[i]) {
            return 0;
        }
        i++;
    }
    if (cadena_a[i] == '\0' && cadena_b[i] == '\0') {
        return 1;
    }
    return 0;
}

    }
}

        char c = cadena[i];

            if (c != 'a' && c != 'e' && c != 'i' && c != 'o' && c != 'u' &&
                c != 'A' && c != 'E' && c != 'I' && c != 'O' && c != 'U') {

            }
        }
    }
    return contador;
}

void kz_decimal_a_binario(int numero, char binario[]) {
    int indice = 0;
    if (numero == 0) {
        binario[0] = '0';
        binario[1] = '\0';
        return;
    }
    while (numero > 0) {
        binario[indice] = (numero % 2) + '0';
        numero = numero / 2;
        indice++;
    }
    binario[indice] = '\0';
    kz_invertir_cadena(binario);
}
